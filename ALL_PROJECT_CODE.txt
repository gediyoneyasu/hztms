import { useState, useEffect } from 'react';
import Modal from '../ui/Modal/Modal.jsx';
import Field from '../ui/Field/Field.jsx';
import Ticket from './Ticket.jsx';
import Toast from '../ui/Toast/Toast.jsx';
import { createBooking } from '../../services/bookingService.js';
import './BookingModal.css';

export default function BookingModal({ schedule, open, onClose }) {
  const [name, setName] = useState('');
  const [phone, setPhone] = useState('');
  const [seats, setSeats] = useState(1);
  const [code, setCode] = useState(null);
  const [loading, setLoading] = useState(false);
  const [showToast, setShowToast] = useState(false);

  const close = () => {
    setName('');
    setPhone('');
    setSeats(1);
    setCode(null);
    onClose();
  };

  const confirm = async () => {
    if (!name.trim() || !phone.trim()) {
      alert('Please enter your name and phone number.');
      return;
    }

    setLoading(true);
    try {
      const res = await createBooking({
        scheduleId: schedule._id,
        passengerName: name,
        phone: phone,
        seats: Number(seats)
      });
      setCode(res.ticketCode);
      setShowToast(true);
      
      // Close toast and modal after 3.5 seconds
      setTimeout(() => {
        setShowToast(false);
        close();
      }, 3500);

    } catch (err) {
      alert('Failed to book: ' + err.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <>
      <Modal open={open} onClose={close}>
        {!code ? (
          <>
            <h3>Confirm your booking</h3>
            <p className="muted booking-meta">
              {schedule?.from} → {schedule?.to} · {schedule?.departure} · ETB {schedule?.price}
            </p>

            <div className="row">
              <Field label="Full name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Your name" />
            </div>
            <div className="row">
              <Field label="Phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="09xx xxx xxx" />
            </div>
            <div className="row">
              <Field label="Seats" type="number" min="1" max="5" value={seats} onChange={(e) => setSeats(e.target.value)} />
            </div>

            <button className="btn btn-primary booking-full" onClick={confirm} disabled={loading}>
              {loading ? 'Processing...' : 'Confirm reservation'}
            </button>
            <button className="btn btn-outline booking-full" onClick={close}>
              Cancel
            </button>
          </>
        ) : (
          <Ticket
            code={code}
            name={name}
            seats={seats}
            schedule={schedule}
            onDone={close}
          />
        )}
      </Modal>

      <Toast 
        message="Booking confirmed! Check your ticket." 
        show={showToast} 
        onClose={() => setShowToast(false)} 
      />
    </>
  );
}

import { useEffect, useRef, useState } from 'react';
import ChatBubble from './ChatBubble.jsx';
import './ChatWidget.css';

const initial = [
  {
    from: 'bot',
    text:
      'Hi! Ask me about routes, fares, departure times or station notices. / ሰላም! ስለ መስመሮች፣ ዋጋ እና የጉዞ ሰዓት ይጠይቁኝ።',
  },
];

export default function ChatWidget() {
  const [open, setOpen] = useState(false);
  const [msgs, setMsgs] = useState(initial);
  const [input, setInput] = useState('');
  const msgsRef = useRef(null);

  useEffect(() => {
    if (msgsRef.current) {
      msgsRef.current.scrollTop = msgsRef.current.scrollHeight;
    }
  }, [msgs, open]);

  const send = (e) => {
    e.preventDefault();
    const text = input.trim();
    if (!text) return;
    setMsgs((prev) => [...prev, { from: 'me', text }]);
    setInput('');

    setTimeout(() => {
      const replies = [
        'Thanks for your message! Our team will respond shortly.',
        'For immediate help, please call +251 46 000 0000.',
        'You can check live schedules on the Schedules page.',
      ];
      const reply = replies[Math.floor(Math.random() * replies.length)];
      setMsgs((prev) => [...prev, { from: 'bot', text: reply }]);
    }, 700);
  };

  return (
    <>
      <button
        className="fab"
        aria-label="Open travel assistant"
        onClick={() => setOpen((v) => !v)}
      >
        💬
      </button>

      {open && (
        <div className="chat">
          <div className="ch">
            <div>
              <b>Travel Assistant</b>
              <div className="ch-sub">English · አማርኛ</div>
            </div>
            <button className="ibtn" onClick={() => setOpen(false)}>✕</button>
          </div>

          <div className="msgs" ref={msgsRef}>
            {msgs.map((m, i) => (
              <ChatBubble key={i} from={m.from} text={m.text} />
            ))}
          </div>

          <form className="chat-form" onSubmit={send}>
            <input
              className="field"
              value={input}
              onChange={(e) => setInput(e.target.value)}
              placeholder="Type your question…"
            />
            <button className="btn btn-dark" type="submit">➤</button>
          </form>
        </div>
      )}
    </>
  );
}

import { useEffect } from 'react';
import './Modal.css';

export default function Modal({ open, onClose, children }) {
  useEffect(() => {
    if (!open) return;
    const handler = (e) => {
      if (e.key === 'Escape') onClose?.();
    };
    document.addEventListener('keydown', handler);
    return () => document.removeEventListener('keydown', handler);
  }, [open, onClose]);

  if (!open) return null;

  return (
    <div
      className="modal-overlay"
      onClick={(e) => {
        if (e.target === e.currentTarget) onClose?.();
      }}
    >
      <div className="modal-box">{children}</div>
    </div>
  );
}

import { createContext, useContext, useState } from 'react';

const AuthContext = createContext(null);

function getStoredUser() {
  const storedUser = localStorage.getItem('hztms-user');
  if (!storedUser) return null;

  try {
    return JSON.parse(storedUser);
  } catch {
    localStorage.removeItem('hztms-user');
    return null;
  }
}

export function AuthProvider({ children }) {
  const [user, setUser] = useState(getStoredUser);

  const login = (userData) => {
    localStorage.setItem('hztms-user', JSON.stringify(userData));
    setUser(userData);
  };

  const logout = () => {
    localStorage.removeItem('hztms-user');
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  return useContext(AuthContext);
}

const LanguageContext = createContext(null);

export function LanguageProvider({ children }) {
  const [lang, setLang] = useState(
    () => localStorage.getItem('hztms-lang') || 'en'
  );

  useEffect(() => {
    localStorage.setItem('hztms-lang', lang);
  }, [lang]);

  const t = (key) => translations[lang]?.[key] ?? key;

  return (
    <LanguageContext.Provider value={{ lang, setLang, t }}>
      {children}
    </LanguageContext.Provider>
  );
}

export function useLanguage() {
  const ctx = useContext(LanguageContext);
  if (!ctx) throw new Error('useLanguage must be used inside LanguageProvider');
  return ctx;
}

import { useEffect, useState } from 'react';
import { API_URL } from '../utils/constants.js';

export function useFetch(endpoint) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    let cancelled = false;
    setLoading(true);

    fetch(`${API_URL}${endpoint}`)
      .then((res) => {
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        return res.json();
      })
      .then((json) => { if (!cancelled) setData(json); })
      .catch((err) => { if (!cancelled) setError(err.message); })
      .finally(() => { if (!cancelled) setLoading(false); });

    return () => { cancelled = true; };
  }, [endpoint]);

  return { data, loading, error };
}

import { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { useLanguage } from '../../../context/LanguageContext.jsx';
import { API_URL } from '../../../utils/constants.js';
import './Hero.css';

export default function Hero() {
  const navigate = useNavigate();
  const { t } = useLanguage();

  const [allSchedules, setAllSchedules] = useState([]);
  const [trip, setTrip] = useState({
    from: '',
    to: '',
    date: new Date().toISOString().slice(0, 10),
  });

  // 1. Fetch data from Admin
  useEffect(() => {
    fetch(`${API_URL}/schedules`)
      .then(res => res.json())
      .then(data => {
        const list = Array.isArray(data) ? data : [];
        setAllSchedules(list);
        
        // Auto-select the first station and destination
        if (list.length > 0) {
          setTrip(prev => ({
            ...prev,
            from: list[0].from || 'Hossana',
            to: list[0].to
          }));
        }
      })
      .catch(err => console.error("Failed to load schedules:", err));
  }, []);

  // 2. Get the exact list of Stations the Admin added
  const availableStations = [...new Set(allSchedules.map(s => s.from || 'Hossana'))];
  
  // 3. Get the Destinations ONLY for the selected Station
  const availableDestinations = [...new Set(
    allSchedules
      .filter(s => (s.from || 'Hossana') === trip.from)
      .map(s => s.to)
  )];

  // 4. If the user changes the Station, automatically update the Destination to the first available one
  useEffect(() => {
    if (availableDestinations.length > 0 && !availableDestinations.includes(trip.to)) {
      setTrip(prev => ({ ...prev, to: availableDestinations[0] }));
    }
  }, [trip.from, availableDestinations, trip.to]);

  const handleSearch = (e) => {
    e.preventDefault();
    if (!trip.from || !trip.to) return;
    navigate(`/booking?from=${trip.from}&to=${trip.to}&date=${trip.date}`);
  };

  const scrollToHow = () => {
    document.querySelector('.how')?.scrollIntoView({ behavior: 'smooth' });
  };

  return (
    <section className="hero-section">
      <div className="hero-container">
        <div className="hero-left">
          <div className="hero-badge">
            <span className="dot"></span> {t('heroBadge')}
          </div>

          <div className="hero-buttons">
            <button className="btn-primary-pill" onClick={() => navigate('/schedules')}>
              {t('btnExplore')} <span>→</span>
            </button>
            <button className="btn-outline-pill" onClick={scrollToHow}>
              <div className="play-icon">▶</div> {t('btnHow')}
            </button>
          </div>
          <div className="hero-stats">
            <div className="stat-item">
              <span className="stat-icon">🚌</span>
              <div><b>250+</b><small>{t('statTrips')}</small></div>
            </div>
            <div className="stat-item">
              <span className="stat-icon">👥</span>
              <div><b>1,200+</b><small>{t('statPassengers')}</small></div>
            </div>
            <div className="stat-item">
              <span className="stat-icon">🏆</span>
              <div><b>15+</b><small>{t('statYears')}</small></div>
            </div>
          </div>
        </div>
        <div className="hero-right">

              ))}

          </div>
          
          {/* DESTINATION DROPDOWN (Filtered) */}
          <div>

      alert(err.message);

      setSaving(false);
    }

    setSaving(true);

        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ ...foundForm, imageUrl })

      setTab('view');

      alert(err.message);
    } finally {
      setSaving(false);
    }

import { API_URL } from '../utils/constants.js';

export async function request(path, options = {}) {
  const res = await fetch(`${API_URL}${path}`, {
    headers: { 'Content-Type': 'application/json', ...(options.headers || {}) },
    ...options,
  });

  if (!res.ok) {
    const body = await res.text();
    throw new Error(body || `Request failed with ${res.status}`);
  }

  return res.json();
}

function resolveApiUrl() {
  const host = typeof window !== 'undefined' ? window.location.hostname : '';
  const onLocalhost = host === 'localhost' || host === '127.0.0.1';
  const configured = import.meta.env.VITE_API_URL;

  if (onLocalhost) return configured || LOCAL_API;
  if (configured && !String(configured).includes('localhost')) return configured;
  return PRODUCTION_API;
}

const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');

// Generate JWT
const generateToken = (id) => {
  return jwt.sign({ id }, process.env.JWT_SECRET, { expiresIn: '30d' });
};

// @desc Register user
const registerUser = async (req, res) => {
  const { name, email, password } = req.body;
  try {
    if (!name || !email || !password) {
      return res.status(400).json({ message: 'Name, email, and password are required' });
    }
    const userExists = await User.findOne({ email });
    if (userExists) return res.status(400).json({ message: 'User already exists' });

    const salt = await bcrypt.genSalt(10);
    const hashedPassword = await bcrypt.hash(password, salt);

    const role = req.body.role === 'Driver' ? 'Driver' : 'Passenger';
    const user = await User.create({ name, email, password: hashedPassword, role });
    res.status(201).json({
      _id: user.id, name: user.name, email: user.email, role: user.role,
      token: generateToken(user._id)
    });
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};

// @desc Login user
const loginUser = async (req, res) => {
  const { email, password } = req.body;
  try {
    const user = await User.findOne({ email });
    if (user && (await bcrypt.compare(password, user.password))) {
      res.json({
        _id: user.id, name: user.name, email: user.email, role: user.role,
        token: generateToken(user._id)
      });
    } else {
      res.status(401).json({ message: 'Invalid credentials' });
    }
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};

module.exports = { registerUser, loginUser };

const jwt = require('jsonwebtoken');
const User = require('../models/User');

const protect = async (req, res, next) => {
  const authorization = req.headers.authorization || '';
  const [scheme, token] = authorization.split(' ');
  if (scheme !== 'Bearer' || !token) {
    return res.status(401).json({ message: 'Authentication required' });
  }

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    const user = await User.findById(decoded.id).select('-password');
    if (!user) return res.status(401).json({ message: 'User account not found' });
    req.user = user;
    return next();
  } catch (error) {
    return res.status(401).json({ message: 'Invalid or expired authentication token' });
  }
};

const admin = (req, res, next) => {
  if (req.user?.role !== 'Admin') {
    return res.status(403).json({ message: 'Administrator access required' });
  }
  return next();
};

module.exports = { protect, admin };

const app = express();

const DEFAULT_ORIGINS = [
  'http://localhost:5173',
  'http://127.0.0.1:5173',
  'http://localhost:4173',
  'https://hztms-frontend.vercel.app',
  'https://hztms.vercel.app',
];

function allowedOrigins() {
  const extra = (process.env.CLIENT_URL || '')
    .split(',')
    .map((value) => value.trim())
    .filter(Boolean);
  return [...new Set([...DEFAULT_ORIGINS, ...extra])];
}

app.use(cors({
  origin(origin, callback) {
    if (!origin) return callback(null, true);
    if (allowedOrigins().includes(origin)) return callback(null, true);
    if (/^https:\/\/hztms(-frontend)?[a-z0-9-]*\.vercel\.app$/i.test(origin)) return callback(null, true);
    return callback(null, false);
  },
}));
app.use(express.json({ limit: '2mb' }));
